
worker_processes 1;

events {
    worker_connections 1024;
}

http {
    include mime.types;
    default_type application/octet-stream;
    sendfile on;
    keepalive_timeout 65;
    server_names_hash_bucket_size 64;

    # Redirect all HTTP traffic to HTTPS
    # HTTP redirection to HTTPS for both www and non-www
server {
    listen 80;
    server_name quadilateral.online www.quadilateral.online;

    return 301 https://www.quadilateral.online$request_uri;
}

# HTTPS configuration for both www and non-www
server {
    listen 443 ssl;
    server_name quadilateral.online www.quadilateral.online;

    ssl_certificate     C:/nginx-1.24.0/certs/cert.pem;
    ssl_certificate_key C:/nginx-1.24.0/certs/key.pem;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # SharedArrayBuffer headers (important for your USD viewer)
    add_header Cross-Origin-Opener-Policy "same-origin" always;
    add_header Cross-Origin-Embedder-Policy "require-corp" always;
    add_header Cross-Origin-Resource-Policy "cross-origin" always;

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    # Special rule for Firebase authentication pages
    location ~ ^/(auth|login|api/auth) {
        proxy_pass http://localhost:3000;
        add_header Cross-Origin-Opener-Policy "unsafe-none" always;
        add_header Cross-Origin-Embedder-Policy "unsafe-none" always;
    }

    # Handling /usd_files/ and adding CORS headers
    location /usd_files/ {
        add_header 'Access-Control-Allow-Origin' '*' always;
        add_header 'Cross-Origin-Resource-Policy' 'cross-origin' always;
        root c:/chat-avatar-app/chat-avatar-app/public;

    location /audio/ {
        add_header 'Access-Control-Allow-Origin' '*' always;
        add_header 'Cross-Origin-Resource-Policy' 'cross-origin' always;
        root c:/chat-avatar-app/chat-avatar-app/public;
    }
}

}
